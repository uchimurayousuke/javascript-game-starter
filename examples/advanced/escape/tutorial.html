比較欄の読み方

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

class RoomObject {
  constructor(x, y, width, height, name) {
    this.x = x;

    this.width = width;
    this.height = height;
    this.name = name;
  }

  contains(x, y) {
    return x >= this.x && x < this.x + this.width &&
      y >= this.y && y < this.y + this.height;
  }

  draw() {
    ctx.fillStyle = "#b7cbd4";
    ctx.fillRect(this.x, this.y, this.width, this.height);

    ctx.fillText(this.name, this.x + 15, this.y + 35);
  }

class Inventory {
  constructor() {
    this.items = [];
    this.selected = "";
  }

  add(item) {
    if (!this.items.includes(item)) this.items.push(item);
  }

  click(x, y) {
    for (let i = 0; i < this.items.length; i++) {
      const left = 30 + i * 150;
      if (x >= left && x < left + 140 && y >= 390 && y < 435) {
        this.selected = this.items[i];
        return true;
      }
    }
    return false;
  }

  draw() {
    for (let i = 0; i < this.items.length; i++) {
      const left = 30 + i * 150;
      ctx.fillStyle = "#f8dc8b";
      ctx.fillRect(left, 390, 140, 45);
      if (this.items[i] === this.selected) {
        ctx.strokeStyle = "#c45631";
        ctx.lineWidth = 4;
        ctx.strokeRect(left, 390, 140, 45);
      }
      ctx.fillStyle = "#24354a";
      ctx.font = "17px sans-serif";
      ctx.fillText(this.items[i], left + 8, 418);
    }
  }
}
