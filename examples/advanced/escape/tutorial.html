比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

コードの意味

start() は始めるときの設定、update() は毎回の動きや判定、draw() は画面の描画です。gameLoop() は update() と draw() を順に呼び、requestAnimationFrame() で次の更新を予約します。再挑戦では動いているゲームループを増やさず、状態を戻して続けます。

class RoomObject は部屋の物を作るための設計図です。constructor(x, y, width, height, name) は new で実体を作るときに呼ばれ、受け取った値を保存します。

this.x = x の左側は今作っている実体自身の位置、右側は渡された値です。本棚と引き出しをそれぞれ new RoomObject(...) で作るので、片方の座標を変えてももう片方は変わりません。

contains(x, y) と draw() は各実体が使えるメソッドです。object.contains(point.x, point.y) はクリック位置がその物の四角に入るか調べ、object.draw() はその物自身の座標と名前で描きます。

getBoundingClientRect() は画面上のCanvasの位置と大きさです。左上からの距離に canvas.width / rect.width などを掛け、縮小表示でもCanvasの座標に直します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

class RoomObject {
  constructor(x, y, width, height, name) {
    this.x = x;
    this.y = y;
    this.width = width;
    this.height = height;
    this.name = name;
  }

  contains(x, y) {
    return x >= this.x && x < this.x + this.width &&
      y >= this.y && y < this.y + this.height;
  }

  draw() {
    ctx.fillStyle = "#b7cbd4";
    ctx.fillRect(this.x, this.y, this.width, this.height);
    ctx.fillStyle = "#24354a";
    ctx.font = "22px sans-serif";
    ctx.fillText(this.name, this.x + 15, this.y + 35);
  }
}

let objects, message;

function start() {
  objects = [
    new RoomObject(60, 105, 160, 180, "本棚"),
    new RoomObject(300, 130, 170, 140, "引き出し"),
    new RoomObject(535, 120, 165, 150, "換気口"),
    new RoomObject(625, 285, 120, 80, "扉")
  ];
  message = "気になる物をクリックしよう";
}

function update() {
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#24354a";
  ctx.font = "22px sans-serif";
  ctx.fillText("部屋の物をクリックして調べよう", 45, 45);
  for (const object of objects) object.draw();
  ctx.fillText(message, 45, 370);
}

function pointFromClick(event) {
  const rect = canvas.getBoundingClientRect();
  return {
    x: (event.clientX - rect.left) * canvas.width / rect.width,
    y: (event.clientY - rect.top) * canvas.height / rect.height
  };
}

canvas.addEventListener("click", (event) => {
  const point = pointFromClick(event);
  for (const object of objects) {
    if (object.contains(point.x, point.y)) {
      message = object.name + "を調べた";
      return;
    }
  }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

コードの意味

class Inventory は持ち物を管理する設計図です。new Inventory() で作った実体は this.items と this.selected を持ちます。this は呼び出された持ち物の実体です。

add(item) は includes() で同じ道具をすでに持っているか調べ、なければ push() します。同じ番号を何度確かめてもドライバーは増えません。

inventory.draw() が持ち物欄を描き、inventory.click(x, y) が押した欄を探して選びます。選ばれた道具だけ枠で示します。部屋に戻ってから欄をクリックします。

コードの意味

引き出しに241を入力すると、inventory.add("ドライバー") で道具を得ます。部屋の持ち物欄で選び、換気口をクリックすると "鍵" を追加します。

鍵を選んで扉をクリックすると state = "clear"。道具なしで換気口や扉を調べても、メッセージが出るだけです。道具を違う場所に使っても消えず、本棚や引き出しに戻れるので、順番を間違えても続けられます。

換気口を開けると ventOpen = true になり、部屋を描くとき換気口にも「開いた」と表示します。引き出しの unlocked と別の状態なので、それぞれ独立して見分けられます。

state !== "playing" ならクリック処理を終えます。Rの start() は物、番号、持ち物、状態を作り直すため、終了後も最初から再挑戦できます。

コードの意味

class Game は1回のゲーム全体をまとめる設計図です。constructor() は部屋の RoomObject たちと new Inventory()、番号、画面、状態をその実体の this に保存します。

game = new Game() は新しいゲームの実体を作ります。Rで start() を呼ぶと、前の実体ではなく新しい実体が入り、持ち物と番号も初期値になります。

game.click(x, y) の中の this はクリックされたゲームの実体です。this.inventory.click(x, y) の中の this は持ち物の実体に切り替わります。どのメソッドを、どの実体に対して呼んだかで this が決まります。

Game.draw() は this.objects の各 object.draw() と this.inventory.draw() を呼びます。this.unlocked と this.ventOpen を見て「開いた」表示を重ね、物の描画、持ち物の描画、ゲーム全体の進行を分けて読めます。

class RoomObject {
  constructor(x, y, width, height, name) {
    this.x = x;

    return x >= this.x && x < this.x + this.width &&
      y >= this.y && y < this.y + this.height;
  }

  draw() {
    ctx.fillStyle = "#b7cbd4";
    ctx.fillRect(this.x, this.y, this.width, this.height);

  }

class Inventory {
  constructor() {
    this.items = [];
    this.selected = "";
  }

  add(item) {
    if (!this.items.includes(item)) this.items.push(item);
  }

  click(x, y) {
    for (let i = 0; i < this.items.length; i++) {
      const left = 30 + i * 150;
      if (x >= left && x < left + 140 && y >= 390 && y < 435) {
        this.selected = this.items[i];
        return true;
      }
    }
    return false;
  }

  draw() {
    for (let i = 0; i < this.items.length; i++) {
      const left = 30 + i * 150;
      ctx.fillStyle = "#f8dc8b";
      ctx.fillRect(left, 390, 140, 45);
      if (this.items[i] === this.selected) {
        ctx.strokeStyle = "#c45631";
        ctx.lineWidth = 4;
        ctx.strokeRect(left, 390, 140, 45);
      }
      ctx.fillStyle = "#24354a";
      ctx.font = "17px sans-serif";
      ctx.fillText(this.items[i], left + 8, 418);
    }
  }
}
