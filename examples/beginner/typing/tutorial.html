コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
    if (position === words[wordIndex].length) {

      position = 0;
      if (wordIndex === words.length) state = "clear";
    }
  }

コードの意味

words = ["a", "cat", "dog"] は複数の文字列を順に入れる配列です。番号は0から始まり、words[0] は a、words[1] は cat です。

wordIndex は今の単語の番号、position はその単語で入力できた文字数です。words[wordIndex][position] で、今の単語の次の1文字を取り出します。

たとえば cat の文字番号は c が0、a が1、t が2です。position が0のとき c なら position++ で1に増え、x なら増えません。最後の t が合うと次の単語へ進み、position は0に戻ります。

length は文字や配列の長さです。cat の length は3。slice(0, position) は先頭から入力できたところまでを取り出し、画面に表示します。

keydown はキーを押したときに動き、event.key がその文字です。toLowerCase() で大文字を小文字に直し、英字1文字だけを比べます。=== は同じか、!== は同じでないかを調べます。全部終わってからRを押すと start() で最初に戻ります。

event.repeat は押しっぱなし、ctrlKey・altKey・metaKey は組み合わせ、isComposing は日本語などの変換中を表します。|| でどれか一つを満たせば return し、その入力を受け取りません。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const words = ["a", "cat", "dog"];
let wordIndex, position, state;

function start() {
  wordIndex = 0;
  position = 0;
  state = "playing";
}

function update() {
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  if (state === "playing") {
    ctx.fillText("表示された英字を順番に打とう", 180, 110);
    ctx.font = "56px sans-serif";
    ctx.fillText(words[wordIndex], 300, 210);
    ctx.font = "24px sans-serif";
    ctx.fillText("打てた文字: " + words[wordIndex].slice(0, position), 230, 280);
  } else {
    ctx.fillText("全部打てた！ R キーでもう一度", 150, 220);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
    if (position === words[wordIndex].length) {
      wordIndex++;
      position = 0;
      if (wordIndex === words.length) state = "clear";
    }
  }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  if (state === "playing") {
    ctx.fillText("表示された英字を順番に打とう", 180, 110);
    ctx.font = "56px sans-serif";
    ctx.fillText(words[wordIndex], 300, 210);
    ctx.font = "24px sans-serif";
    ctx.fillText("打てた文字: " + words[wordIndex].slice(0, position), 230, 280);
  } else {
    ctx.fillText("全部打てた！ R キーでもう一度", 150, 220);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
    if (position === words[wordIndex].length) {
      wordIndex++;
      position = 0;
      if (wordIndex === words.length) state = "clear";
    }
  }
});

変更後（コピー用）：

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("得点: " + score + "  まちがい: " + mistakes + "  残り: " + seconds + "秒", 100, 65);
  if (state === "playing") {
    ctx.fillText("英字を順番に打とう", 260, 120);
    ctx.font = "56px sans-serif";
    ctx.fillText(words[wordIndex], 300, 210);
    ctx.font = "24px sans-serif";
    ctx.fillText("打てた文字: " + words[wordIndex].slice(0, position), 230, 270);
    ctx.fillText("1語完成: 1文字10点 / 正確 +5 / 3秒未満 +5", 110, 335);
  } else {
    if (state === "clear") {
      ctx.fillText("全部打てた！", 310, 200);
    } else {
      ctx.fillText("時間切れ！", 320, 200);
    }
    ctx.fillText("R キーでもう一度", 270, 270);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (performance.now() >= endTime) {
    seconds = 0;
    state = "over";
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
    if (position === words[wordIndex].length) {
      score += words[wordIndex].length * 10;
      if (wordErrors === 0) score += 5;
      if (performance.now() - wordStarted < 3000) score += 5;
      wordIndex++;
      position = 0;
      wordErrors = 0;
      wordStarted = performance.now();
      if (wordIndex === words.length) {
        seconds = Math.max(0, Math.ceil((endTime - performance.now()) / 1000));
        state = "clear";
      }
    }
  } else {
    mistakes++;
    wordErrors++;
  }
});

コードの意味

performance.now() は時間をミリ秒で返します。1000ミリ秒が1秒です。開始時の時刻に30000を足した endTime が、30秒後の締切です。

endTime - performance.now() は残りのミリ秒です。1000で割り、Math.ceil で端数を切り上げ、Math.max(0, ...) で0より小さくしません。残り秒数は seconds に入れ、終了後は表示も止めます。

wordStarted は今の単語を表示した時刻です。今の時刻から引くと経過時間になります。3000ミリ秒より短ければ速さの5点を加え、ちょうど3000では加えません。

単語が完成すると、score += words[wordIndex].length * 10 で1文字につき10点を加えます。wordErrors === 0 なら間違いなしの5点も加えます。cat は基本30点で、両方の加点があれば40点です。

mistakes は全体の間違い数、wordErrors は今の単語の間違い数です。次の単語に進むと wordErrors と position を0にし、wordStarted を新しい時刻にします。

state が "playing" の間だけ入力と時間を進めます。全部打つと "clear"、締切になると "over"。Rで start() を呼び、得点と時間も初期化します。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const words = ["a", "cat", "dog", "sun", "fish"];
let wordIndex, position, state, score, mistakes, wordErrors, wordStarted, endTime, seconds;

function start() {
  wordIndex = 0;
  position = 0;
  state = "playing";
  score = 0;
  mistakes = 0;
  wordErrors = 0;
  wordStarted = performance.now();
  endTime = wordStarted + 30000;
  seconds = 30;
}

function update() {
  if (state === "playing") {
    seconds = Math.max(0, Math.ceil((endTime - performance.now()) / 1000));
    if (seconds === 0) state = "over";
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("得点: " + score + "  まちがい: " + mistakes + "  残り: " + seconds + "秒", 100, 65);
  if (state === "playing") {
    ctx.fillText("英字を順番に打とう", 260, 120);
    ctx.font = "56px sans-serif";
    ctx.fillText(words[wordIndex], 300, 210);
    ctx.font = "24px sans-serif";
    ctx.fillText("打てた文字: " + words[wordIndex].slice(0, position), 230, 270);
    ctx.fillText("1語完成: 1文字10点 / 正確 +5 / 3秒未満 +5", 110, 335);
  } else {
    if (state === "clear") {
      ctx.fillText("全部打てた！", 310, 200);
    } else {
      ctx.fillText("時間切れ！", 320, 200);
    }
    ctx.fillText("R キーでもう一度", 270, 270);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (performance.now() >= endTime) {
    seconds = 0;
    state = "over";
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
    if (position === words[wordIndex].length) {
      score += words[wordIndex].length * 10;
      if (wordErrors === 0) score += 5;
      if (performance.now() - wordStarted < 3000) score += 5;
      wordIndex++;
      position = 0;
      wordErrors = 0;
      wordStarted = performance.now();
      if (wordIndex === words.length) {
        seconds = Math.max(0, Math.ceil((endTime - performance.now()) / 1000));
        state = "clear";
      }
    }
  } else {
    mistakes++;
    wordErrors++;
  }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();
