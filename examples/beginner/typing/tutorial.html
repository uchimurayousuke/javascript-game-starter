コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

入力を受け取る

キーを押したときの処理を登録します。

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;

keydown はキーを押したときに起きます。event.repeat は押しっぱなし、ctrlKey・altKey・metaKey は組み合わせ、isComposing は日本語などの変換中を表します。|| でいずれかに当てはまれば return で処理を終えます。toLowerCase() で入力を小文字にします。

終了後のやり直しと英字の確認

プレイ中以外は R キーで初期状態に戻します。

if (state !== "playing") {
  if (key === "r") start();
  return;
}
if (key.length !== 1 || key < "a" || key > "z") return;

!== は同じでないか、=== は同じかを調べます。終了後は R なら start() で戻し、ほかの入力もそこで終えます。プレイ中は英字1文字以外を受け取りません。動いている gameLoop() は再び呼びません。二重に動くのを防ぐためです。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const words = ["a", "cat", "dog", "sun", "fish"];
let wordIndex, position, state, score, mistakes, wordErrors, wordStarted, endTime, seconds;

function start() {
  wordIndex = 0;
  position = 0;
  state = "playing";
  score = 0;
  mistakes = 0;
  wordErrors = 0;
  wordStarted = performance.now();
  endTime = wordStarted + 30000;
  seconds = 30;
}

function update() {
  if (state === "playing") {
    seconds = Math.max(0, Math.ceil((endTime - performance.now()) / 1000));
    if (seconds === 0) state = "over";
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("得点: " + score + "  まちがい: " + mistakes + "  残り: " + seconds + "秒", 100, 65);
  if (state === "playing") {
    ctx.fillText("英字を順番に打とう", 260, 120);
    ctx.font = "56px sans-serif";
    ctx.fillText(words[wordIndex], 300, 210);
    ctx.font = "24px sans-serif";
    ctx.fillText("打てた文字: " + words[wordIndex].slice(0, position), 230, 270);
    ctx.fillText("1語完成: 1文字10点 / 正確 +5 / 3秒未満 +5", 110, 335);
  } else {
    if (state === "clear") {
      ctx.fillText("全部打てた！", 310, 200);
    } else {
      ctx.fillText("時間切れ！", 320, 200);
    }
    ctx.fillText("R キーでもう一度", 270, 270);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  const key = event.key.toLowerCase();
  if (state !== "playing") {
    if (key === "r") start();
    return;
  }
  if (performance.now() >= endTime) {
    seconds = 0;
    state = "over";
    return;
  }
  if (key.length !== 1 || key < "a" || key > "z") return;
  if (key === words[wordIndex][position]) {
    position++;
