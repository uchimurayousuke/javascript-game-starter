タイミングストップ

初心者向け

左右に動く青い線をスペースキーで止めます。線の中心が緑の範囲に入れば成功です。

完成版で遊ぶ

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

始める前に

template/ の index.html、style.css、script.js を新しい作業用フォルダーにコピーし、同じ場所に置きます。元のひな型は残してください。

各ステップでは作業用の script.js を開き、示した部分を順番に置き換えます。index.html と style.css は変更しません。保存後、作業用の index.html を開くか再読み込みします。

document.querySelector("#game-canvas") はHTMLのCanvasを選び、getContext("2d") で絵を描く道具を取り出します。start() は最初の設定、update() は更新、draw() は描画です。gameLoop() は requestAnimationFrame で更新と描画を繰り返します。

コードの意味

start() は始めるときの設定、update() は毎回の動きや判定、draw() は画面の描画です。gameLoop() は update() と draw() を順に呼び、requestAnimationFrame() で次の更新を予約します。再挑戦では動いているゲームループを増やさず、状態を戻して続けます。

let markerX は、青い線の中心の横の位置を覚える変数です。start() の markerX = 80 で最初の位置を決めます。

fillRect(300, 200, 200, 24) は、左上が横300・縦200で、幅200・高さ24の緑の帯を描きます。右端は300 + 200で500です。

青い線の幅は10です。markerX - 5 は中心から幅の半分だけ左へ戻り、四角の左端を求めます。たとえば中心が80なら左端は75です。

fillStyle は色、font は文字の大きさ、fillText は文字と表示する位置を決めます。clearRect は前の絵を消します。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let markerX;

function start() {
  markerX = 80;
}

function update() {
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#d9e8f9";
  ctx.fillRect(80, 200, 640, 24);
  ctx.fillStyle = "#50a76b";
  ctx.fillRect(300, 200, 200, 24);
  ctx.fillStyle = "#204a84";
  ctx.fillRect(markerX - 5, 180, 10, 64);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("青い線を緑の中で止めよう", 190, 120);
}

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

speed は1回の更新で進む量です。markerX += speed は、今の位置に速さを足して markerX に戻す書き方です。位置100で速さ4なら次は104です。

markerX >= 720 || markerX <= 80 の || は「または」です。右端720以上、または左端80以下なら if の中へ進みます。

speed = -speed は速さの符号を反転します。右向きの4は左向きの-4になり、-4は4に戻ります。次の更新では逆向きに動きます。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let markerX, speed;

function start() {
  markerX = 80;
  speed = 4;
}

function update() {
  markerX += speed;
  if (markerX >= 720 || markerX <= 80) {
    speed = -speed;
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#d9e8f9";
  ctx.fillRect(80, 200, 640, 24);
  ctx.fillStyle = "#50a76b";
  ctx.fillRect(300, 200, 200, 24);
  ctx.fillStyle = "#204a84";
  ctx.fillRect(markerX - 5, 180, 10, 64);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("青い線が左右に動きます", 220, 120);
}

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

  ctx.fillStyle = "#204a84";
  ctx.fillRect(markerX - 5, 180, 10, 64);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("スペースで止める", 270, 120);
  if (state === "stopped") {
    if (success) {
      ctx.fillText("成功！ 緑の中で止まった", 220, 300);
    } else {
      ctx.fillText("おしい！ 緑の外で止まった", 205, 300);
    }
    ctx.fillText("R キーでもう一度", 260, 350);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space") {
    event.preventDefault();
    if (state === "moving") {
      state = "stopped";
      success = markerX >= 300 && markerX <= 500;
    }
  } else if (event.key.toLowerCase() === "r" && state === "stopped") {
    start();
  }
});

コードの意味

state は動作の状態を文字列で覚えます。"moving" は動いている状態、"stopped" は止まった状態です。=== は左右が同じか調べ、update() は state === "moving" の間だけ動かします。

success は成功かどうかを表す変数です。true は成功、false は失敗です。再開時は start() で位置、速さ、状態、成功の記録を初期化します。

markerX >= 300 && markerX <= 500 の && は「両方」です。中心が300以上で、かつ500以下なら true。500は成功、501は範囲外です。幅全体ではなく線の中心を比べます。

window.addEventListener("keydown", ...) は、キーが押されたときの処理を登録します。event.code === "Space" でスペースキーを調べ、preventDefault() でページのスクロールを止めます。

event.repeat は押しっぱなしの繰り返しです。ctrlKey・altKey・metaKey は他のキーとの組み合わせ、isComposing は日本語などの変換中を表します。どれかが当てはまると return でその回の処理を終えます。

else if は前の条件が違う場合に次の条件を調べます。event.key.toLowerCase() は英字を小文字にして、止まった後のRキーを受け取ります。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let markerX, speed, state, success;

function start() {
  markerX = 80;
  speed = 4;
  state = "moving";
  success = false;
}

function update() {
  if (state === "moving") {
    markerX += speed;
    if (markerX >= 720 || markerX <= 80) {
      speed = -speed;
    }
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#d9e8f9";
  ctx.fillRect(80, 200, 640, 24);
  ctx.fillStyle = "#50a76b";
  ctx.fillRect(300, 200, 200, 24);
  ctx.fillStyle = "#204a84";
  ctx.fillRect(markerX - 5, 180, 10, 64);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("スペースで止める", 270, 120);
  if (state === "stopped") {
    if (success) {
      ctx.fillText("成功！ 緑の中で止まった", 220, 300);
    } else {
      ctx.fillText("おしい！ 緑の外で止まった", 205, 300);
    }
    ctx.fillText("R キーでもう一度", 260, 350);
  }
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space") {
    event.preventDefault();

function gameLoop() {
