ブロックくずしの発展

基本の完成版から3つの機能を順番に追加します。

発展編の完成版で遊ぶ

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

始める前に

基本の完成版の index.html、style.css、script.js を新しい作業用フォルダーにコピーし、同じ場所に置きます。自分で作った基本版でも同じ内容なら使えます。

各段階の変更を上から順に作業用の script.js に加えます。変更前と同じ範囲を探して、変更後のコードに置き換えてください。保存したら index.html を開くか、再読み込みします。

コードの意味

各ブロックの hp は残りの耐久です。列の番号が4の倍数なら2、それ以外は1から始まります。

当たると hp-- で1減らし、0のときだけ消して得点を増やします。色の濃いブロックは2回の別々の接触が必要です。

反射後はボールをブロックの外へ戻し、locked を立てます。ボールがブロックの並ぶ高さから離れるまで、同じブロックの接触を再び数えません。

この段階の script.js 全文を確認する
"use strict";
const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, y, dx, dy, paddleX;
const radius = 10, paddleWidth = 110, paddleY = 410;
let left = false, right = false, bricks, score, state;

function start() {
  x = 400; y = 225; dx = 4; dy = 4; paddleX = 345;
  left = false; right = false; score = 0; state = "playing";
  bricks = [];
  for (let row = 0; row < 3; row++) {
    for (let col = 0; col < 8; col++) {
      bricks.push({ x: 80 + col * 80, y: 55 + row * 30, alive: true, hp: col % 4 === 0 ? 2 : 1, locked: false });
    }
  }
}

function update() {
  if (state !== "playing") return;
  if (left) paddleX -= 7;
  if (right) paddleX += 7;
  paddleX = Math.max(0, Math.min(canvas.width - paddleWidth, paddleX));
  x += dx; y += dy;
  if (x <= radius || x >= canvas.width - radius) dx = -dx;
  if (y <= radius) dy = -dy;
  if (dy > 0 && y + radius >= paddleY && y - radius <= paddleY + 14 &&
      x + radius >= paddleX && x - radius <= paddleX + paddleWidth) {
    y = paddleY - radius; dy = -dy;
  }
  if (bricks.every(brick => y + radius < brick.y || y - radius > brick.y + 20)) {
    for (const brick of bricks) brick.locked = false;
  }
  for (const brick of bricks) {
    if (brick.alive && !brick.locked && x + radius >= brick.x && x - radius <= brick.x + 70 &&
        y + radius >= brick.y && y - radius <= brick.y + 20) {
      brick.hp--;
      if (brick.hp === 0) { brick.alive = false; score++; }
      dy = -dy;
      if (dy < 0) y = brick.y - radius - 0.01;
      else y = brick.y + 20 + radius + 0.01;
      brick.locked = true;
      break;
    }
  }
  if (score === bricks.length) state = "clear";
  if (y - radius > canvas.height) state = "over";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#2269b4";
  ctx.beginPath(); ctx.arc(x, y, radius, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#26334a";
  ctx.fillRect(paddleX, paddleY, paddleWidth, 14);
  for (const brick of bricks) {
    if (brick.alive) {
      ctx.fillStyle = brick.hp === 2 ? "#a64d42" : "#ec8055";
      ctx.fillRect(brick.x, brick.y, 70, 20);
    }
  }
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  ctx.fillText("こわした数: " + score + " / " + bricks.length, 20, 35);
  if (state === "clear") ctx.fillText("クリア！ R キーでもう一度", 210, 225);
  if (state === "over") ctx.fillText("ゲームオーバー！ R キーでもう一度", 180, 225);
}

window.addEventListener("keydown", (event) => {
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") event.preventDefault();
  if (event.key === "ArrowLeft") left = true;
  if (event.key === "ArrowRight") right = true;
});
window.addEventListener("keyup", (event) => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;
});
window.addEventListener("blur", () => { left = false; right = false; });
window.addEventListener("keydown", (event) => {
  if (event.key.toLowerCase() === "r") start();
});
function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

let left = false, right = false, bricks, score, state, lives;

function start() {
  x = 400; y = 225; dx = 4; dy = 4; paddleX = 345;
  left = false; right = false; score = 0; state = "playing"; lives = 3;
  bricks = [];
  for (let row = 0; row < 3; row++) {
    for (let col = 0; col < 8; col++) {
      bricks.push({ x: 80 + col * 80, y: 55 + row * 30, alive: true, hp: col % 4 === 0 ? 2 : 1, locked: false });
    }
  }
}

  if (state !== "playing") return;
  if (left) paddleX -= 7;
  if (right) paddleX += 7;
  paddleX = Math.max(0, Math.min(canvas.width - paddleWidth, paddleX));
  x += dx; y += dy;
  if (x <= radius || x >= canvas.width - radius) dx = -dx;
  if (y <= radius) dy = -dy;
  if (dy > 0 && y + radius >= paddleY && y - radius <= paddleY + 14 &&
      x + radius >= paddleX && x - radius <= paddleX + paddleWidth) {
    y = paddleY - radius; dy = -dy;
  }
  if (bricks.every(brick => y + radius < brick.y || y - radius > brick.y + 20)) {
    for (const brick of bricks) brick.locked = false;

  for (const brick of bricks) {
    if (brick.alive && !brick.locked && x + radius >= brick.x && x - radius <= brick.x + 70 &&

      brick.hp--;

コードの意味

makeBricks() はブロックを作り直す関数です。level に応じて丈夫な列をずらします。

bricks.every(brick => !brick.alive) は、すべてのブロックが消えたかを調べます。1面目なら2面目を作り、2面目ならクリアです。

板に当たった位置のずれを板の半分の幅で割り、5倍して横速度にします。中心より右なら右へ、左なら左へ進みます。0.5なら横速度2.5です。絶対値が1未満なら1か−1にし、真上だけを往復しないようにします。

ステージが変わっても得点と残り回数を持ち越し、ボールと直前の接触は初期位置に戻します。

"use strict";
const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, y, dx, dy, paddleX;
const radius = 10, paddleWidth = 110, paddleY = 410;
let left = false, right = false, bricks, score, state, lives, level;

function start() {
  x = 400; y = 225; dx = 4; dy = 4; paddleX = 345;
  left = false; right = false; score = 0; state = "playing"; lives = 3; level = 1;
  makeBricks();
}

function makeBricks() {
  bricks = [];
  for (let row = 0; row < 3; row++) {
    for (let col = 0; col < 8; col++) {
      bricks.push({ x: 80 + col * 80, y: 55 + row * 30, alive: true, hp: (col + level - 1) % 4 === 0 ? 2 : 1, locked: false });
    }
  }
}

function update() {
  if (state !== "playing") return;
  if (left) paddleX -= 7;
  if (right) paddleX += 7;
  paddleX = Math.max(0, Math.min(canvas.width - paddleWidth, paddleX));
  x += dx; y += dy;
  if (x <= radius || x >= canvas.width - radius) dx = -dx;
  if (y <= radius) dy = -dy;
  if (dy > 0 && y + radius >= paddleY && y - radius <= paddleY + 14 &&
      x + radius >= paddleX && x - radius <= paddleX + paddleWidth) {
    y = paddleY - radius; dy = -dy;
    const offset = (x - (paddleX + paddleWidth / 2)) / (paddleWidth / 2);
    dx = Math.max(-5, Math.min(5, offset * 5));
    if (Math.abs(dx) < 1) dx = dx < 0 ? -1 : 1;
  }
  if (bricks.every(brick => y + radius < brick.y || y - radius > brick.y + 20)) {
    for (const brick of bricks) brick.locked = false;
  }
  for (const brick of bricks) {
    if (brick.alive && !brick.locked && x + radius >= brick.x && x - radius <= brick.x + 70 &&
        y + radius >= brick.y && y - radius <= brick.y + 20) {
      brick.hp--;
      if (brick.hp === 0) { brick.alive = false; score++; }
      dy = -dy;
      if (dy < 0) y = brick.y - radius - 0.01;
      else y = brick.y + 20 + radius + 0.01;
      brick.locked = true;
      break;
    }
  }
  if (bricks.every(brick => !brick.alive)) {
    if (level === 1) {
      level = 2;
      makeBricks();
      x = 400; y = 225; dx = 4; dy = 4;
    } else {
      state = "clear";
    }
  }
  if (y - radius > canvas.height) {
    lives--;
    if (lives === 0) state = "over";
    else { x = 400; y = 225; dx = 4; dy = 4; for (const brick of bricks) brick.locked = false; }
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#2269b4";
  ctx.beginPath(); ctx.arc(x, y, radius, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#26334a";
  ctx.fillRect(paddleX, paddleY, paddleWidth, 14);
  for (const brick of bricks) {
    if (brick.alive) {
      ctx.fillStyle = brick.hp === 2 ? "#a64d42" : "#ec8055";
      ctx.fillRect(brick.x, brick.y, 70, 20);
    }
  }
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  ctx.fillText("ステージ " + level + " / 2  こわした数: " + score + "  残り: " + lives, 20, 35);
  if (state === "clear") ctx.fillText("クリア！ R キーでもう一度", 210, 225);
  if (state === "over") ctx.fillText("ゲームオーバー！ R キーでもう一度", 180, 225);
}

window.addEventListener("keydown", (event) => {
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") event.preventDefault();
  if (event.key === "ArrowLeft") left = true;
  if (event.key === "ArrowRight") right = true;
});
window.addEventListener("keyup", (event) => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;
});
window.addEventListener("blur", () => { left = false; right = false; });
window.addEventListener("keydown", (event) => {
  if (event.key.toLowerCase() === "r") start();
});
function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();
