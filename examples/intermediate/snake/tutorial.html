コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

始める前に

template/ の index.html、style.css、script.js を新しい作業用フォルダーにコピーし、同じ場所に置きます。元のひな型は残してください。

各ステップでは作業用の script.js を開き、示した部分を順番に置き換えます。index.html と style.css は変更しません。保存後、作業用の index.html を開くか再読み込みします。

document.querySelector("#game-canvas") はHTMLのCanvasを選び、getContext("2d") で絵を描く道具を取り出します。start() は最初の設定、update() は更新、draw() は描画です。gameLoop() は requestAnimationFrame で更新と描画を繰り返します。

コードの意味

start() は始めるときの設定、update() は毎回の動きや判定、draw() は画面の描画です。gameLoop() は update() と draw() を順に呼び、requestAnimationFrame() で次の更新を予約します。再挑戦では動いているゲームループを増やさず、状態を戻して続けます。

snake[0] は先頭の頭です。体を描いた後に、そのマスだけ濃い緑で重ね、白い目を付けます。画面の「濃い緑が頭」も目印です。

snake は体の位置を順に入れる配列です。最初の {x: 4, y: 4} が頭、次が (3, 4)、最後が (2, 4) のしっぽです。{} の中に名前と値をまとめたものをオブジェクトと呼びます。

ここでの x と y はマスの番号です。1マスは cell = 30 ピクセル。boardX + part.x * cell と boardY + part.y * cell で、マスの番号をCanvasの位置に直します。

外側と内側の for で行と列を順に調べ、strokeRect でマスの枠を描きます。for (const part of snake) は配列から体の位置を1つずつ取り出します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const cell = 30, boardX = 80, boardY = 70;
let snake;

function start() {
  snake = [{x: 4, y: 4}, {x: 3, y: 4}, {x: 2, y: 4}];
}

function update() {
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.strokeStyle = "#c6d2df";
  for (let row = 0; row < 10; row++) {
    for (let col = 0; col < 10; col++) {
      ctx.strokeRect(boardX + col * cell, boardY + row * cell, cell, cell);
    }
  }
  ctx.fillStyle = "#2d8c58";
  for (const part of snake) {
    ctx.fillRect(boardX + part.x * cell + 1, boardY + part.y * cell + 1, cell - 2, cell - 2);
  }
  ctx.fillStyle = "#145a38";
  ctx.fillRect(boardX + snake[0].x * cell + 1, boardY + snake[0].y * cell + 1, cell - 2, cell - 2);
  ctx.fillStyle = "#ffffff";
  ctx.fillRect(boardX + snake[0].x * cell + 8, boardY + snake[0].y * cell + 9, 4, 4);
  ctx.fillRect(boardX + snake[0].x * cell + 18, boardY + snake[0].y * cell + 9, 4, 4);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("濃い緑が頭", 430, 300);
  ctx.fillText("緑のマスがヘビ", 430, 150);
}

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.key === "ArrowUp" && direction.y !== 1) nextDirection = {x: 0, y: -1};
  if (event.key === "ArrowDown" && direction.y !== -1) nextDirection = {x: 0, y: 1};
  if (event.key === "ArrowLeft" && direction.x !== 1) nextDirection = {x: -1, y: 0};

コードの意味

direction は現在の進む向き、nextDirection は次に進む向きです。右向きの {x: 1, y: 0} は、横に1マス、縦には0マス進みます。

performance.now() はミリ秒の時刻です。前の移動から200ミリ秒未満なら return で更新を終えます。1000ミリ秒が1秒なので、200ミリ秒は0.2秒です。

頭 (4, 4)、次 (3, 4)、しっぽ (2, 4) が右へ進むと、新しい頭は (5, 4) です。unshift(head) で先頭に加え、pop() で最後を除くので長さは3のままです。

矢印キーは nextDirection を変えます。direction.x !== 1 などで現在の進行方向と真逆を拒み、1回の移動前に何度キーを押しても体へ急に戻れないようにします。この段階だけ、端を越えたら反対側へ出ます。

event.repeat は押しっぱなし、Ctrl・Alt・Metaは組み合わせ、isComposing は変換中です。これらの入力は受け取らず、矢印キーのページ移動は preventDefault() で止めます。event.key.startsWith("Arrow") はキー名が Arrow で始まるかを調べ、矢印キーをまとめて見分けます。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const cell = 30, boardX = 80, boardY = 70;
let snake, direction, nextDirection, lastMove;

function start() {
  snake = [{x: 4, y: 4}, {x: 3, y: 4}, {x: 2, y: 4}];
  direction = {x: 1, y: 0};
  nextDirection = {x: 1, y: 0};
  lastMove = performance.now();
}

function update() {
  if (performance.now() - lastMove < 200) return;
  lastMove = performance.now();
  direction = nextDirection;
  const head = {x: snake[0].x + direction.x, y: snake[0].y + direction.y};
  if (head.x < 0) head.x = 9;
  if (head.x > 9) head.x = 0;
  if (head.y < 0) head.y = 9;
  if (head.y > 9) head.y = 0;
  snake.unshift(head);
  snake.pop();
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.strokeStyle = "#c6d2df";
  for (let row = 0; row < 10; row++) {
    for (let col = 0; col < 10; col++) {
      ctx.strokeRect(boardX + col * cell, boardY + row * cell, cell, cell);
    }
  }
  ctx.fillStyle = "#2d8c58";
  for (const part of snake) {
    ctx.fillRect(boardX + part.x * cell + 1, boardY + part.y * cell + 1, cell - 2, cell - 2);
  }
  ctx.fillStyle = "#145a38";
  ctx.fillRect(boardX + snake[0].x * cell + 1, boardY + snake[0].y * cell + 1, cell - 2, cell - 2);
  ctx.fillStyle = "#ffffff";
  ctx.fillRect(boardX + snake[0].x * cell + 8, boardY + snake[0].y * cell + 9, 4, 4);
  ctx.fillRect(boardX + snake[0].x * cell + 18, boardY + snake[0].y * cell + 9, 4, 4);
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("濃い緑が頭", 430, 300);
  ctx.fillText("矢印キーで進もう", 430, 150);
  ctx.fillText("端から反対側に出ます", 430, 200);
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.key === "ArrowUp" && direction.y !== 1) nextDirection = {x: 0, y: -1};
  if (event.key === "ArrowDown" && direction.y !== -1) nextDirection = {x: 0, y: 1};
  if (event.key === "ArrowLeft" && direction.x !== 1) nextDirection = {x: -1, y: 0};
  if (event.key === "ArrowRight" && direction.x !== -1) nextDirection = {x: 1, y: 0};
  if (event.key.startsWith("Arrow")) event.preventDefault();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

    for (let col = 0; col < 10; col++) {
      let occupied = false;
      for (const part of snake) {
        if (part.x === col && part.y === row) occupied = true;

コードの意味

state が "playing" の間だけ進みます。頭の x または y が0〜9の外なら "over" にし、その回の処理を return で終えます。

eating は新しい頭と食べ物の x・y が両方同じなら true。頭 (4, 4) が右の食べ物 (5, 4) へ進むと、頭を加えてもしっぽを消さず、長さが3から4になります。

食べ物がない普通の移動では、最後のマスは pop() で空きます。そのため体との比較でも、移動後に消える最後のマスだけ continue で飛ばします。食べるときは最後のマスも残るので比べます。

placeFood() は2重の for で全マスを調べ、ヘビがいない場所だけ empty.push({x: col, y: row}) で配列に入れます。Math.random() と Math.floor() でその空きマスを1つ選ぶため、食べ物は体の中に出ません。

!occupied は「占有されていない」、empty.length は空きマスの数です。空きが0ならクリアにします。食べ物5個で "clear"、終了後のRは start() で体、向き、得点、時刻を戻します。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const cell = 30, boardX = 80, boardY = 70;
let snake, direction, nextDirection, lastMove, food, score, state;

function start() {
  snake = [{x: 4, y: 4}, {x: 3, y: 4}, {x: 2, y: 4}];
  direction = {x: 1, y: 0};
  nextDirection = {x: 1, y: 0};
  lastMove = performance.now();
  score = 0;
  state = "playing";
  placeFood();
}

function update() {
  if (state !== "playing") return;
  if (performance.now() - lastMove < 200) return;
  lastMove = performance.now();
  direction = nextDirection;
  const head = {x: snake[0].x + direction.x, y: snake[0].y + direction.y};
  if (head.x < 0 || head.x > 9 || head.y < 0 || head.y > 9) {
    state = "over";
    return;
  }
  const eating = head.x === food.x && head.y === food.y;
  for (let i = 0; i < snake.length; i++) {
    if (!eating && i === snake.length - 1) continue;
    if (snake[i].x === head.x && snake[i].y === head.y) {
      state = "over";
      return;
    }
  }
  snake.unshift(head);
  if (eating) {
    score++;
    if (score >= 5) {
      state = "clear";
    } else {
      placeFood();
    }
  } else {
    snake.pop();
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.strokeStyle = "#c6d2df";
  for (let row = 0; row < 10; row++) {
    for (let col = 0; col < 10; col++) {
      ctx.strokeRect(boardX + col * cell, boardY + row * cell, cell, cell);
    }
  }
  ctx.fillStyle = "#2d8c58";
  for (const part of snake) {
    ctx.fillRect(boardX + part.x * cell + 1, boardY + part.y * cell + 1, cell - 2, cell - 2);
  }
  ctx.fillStyle = "#145a38";
  ctx.fillRect(boardX + snake[0].x * cell + 1, boardY + snake[0].y * cell + 1, cell - 2, cell - 2);
  ctx.fillStyle = "#ffffff";
  ctx.fillRect(boardX + snake[0].x * cell + 8, boardY + snake[0].y * cell + 9, 4, 4);
  ctx.fillRect(boardX + snake[0].x * cell + 18, boardY + snake[0].y * cell + 9, 4, 4);
  if (food !== null) {
    ctx.fillStyle = "#dc5544";
    ctx.fillRect(boardX + food.x * cell + 4, boardY + food.y * cell + 4, cell - 8, cell - 8);
  }
  ctx.fillStyle = "#26334a";
  ctx.font = "24px sans-serif";
  ctx.fillText("濃い緑が頭", 430, 300);
  ctx.fillText("得点: " + score + " / 5", 430, 130);
  if (state === "playing") {
    ctx.fillText("矢印キーで食べ物へ", 430, 190);
  } else {
    if (state === "clear") {
      ctx.fillText("クリア！", 430, 190);
    } else {
      ctx.fillText("ぶつかった！", 430, 190);
    }
    ctx.fillText("R キーでもう一度", 430, 240);
  }
}

function placeFood() {
  const empty = [];
  for (let row = 0; row < 10; row++) {
    for (let col = 0; col < 10; col++) {
      let occupied = false;
      for (const part of snake) {
        if (part.x === col && part.y === row) occupied = true;
      }
      if (!occupied) empty.push({x: col, y: row});
    }
  }
  if (empty.length === 0) {
    state = "clear";
    food = null;
    return;
  }
  food = empty[Math.floor(Math.random() * empty.length)];
}

window.addEventListener("keydown", (event) => {
  if (event.repeat || event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (state !== "playing") {
    if (event.key.toLowerCase() === "r") start();
    return;
  }
  if (event.key === "ArrowUp" && direction.y !== 1) nextDirection = {x: 0, y: -1};
  if (event.key === "ArrowDown" && direction.y !== -1) nextDirection = {x: 0, y: 1};
  if (event.key === "ArrowLeft" && direction.x !== 1) nextDirection = {x: -1, y: 0};
  if (event.key === "ArrowRight" && direction.x !== -1) nextDirection = {x: 1, y: 0};
  if (event.key.startsWith("Arrow")) event.preventDefault();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();
