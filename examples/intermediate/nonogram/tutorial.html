}
function runs(values) {
  const result = []; let count = 0;
  for (const value of [...values, 0]) {
    if (value) count++;
    else if (count) { result.push(count); count = 0; }
  }
  return result.length ? result.join(" ") : "0";

runs() は並びを順に見て、1が続く数を数えます。0で区切って配列に入れ、最後に join(" ") でつなぎます。たとえば1、1、0、1なら「2 1」です。

    ctx.fillText(runs(puzzles[puzzle].map(row => row[i])), ox + i * size + 8, 85);

各行をそのまま runs() に渡します。列は map() で各行の同じ位置を集めてから渡します。行は左側、列は上側に表示します。
