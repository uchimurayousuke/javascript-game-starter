コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let state, x, startedAt;

function start() {
  state = "playing"; x = 100; startedAt = performance.now();
}

function update() {
  if (state !== "playing") return;
  x += 2; if (x > 700) x = 100;
  if (performance.now() - startedAt >= 10000) state = "over";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "28px sans-serif";
  ctx.fillText("残り " + Math.max(0, Math.ceil((10000 - (performance.now() - startedAt)) / 1000)) + "秒", 60, 80);
  ctx.fillRect(x, 200, 40, 40);
  ctx.fillText(state === "paused" ? "停止中　P: 再開" : state === "over" ? "時間切れ　R: 再開" : "動いている", 200, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

}

function update() {
  if (state !== "playing") return;

P で停止するとき現在時刻を pausedAt に記録します。再開時にはその時刻からの差を pausedTotal に加えてから "playing" に戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let state, x, startedAt, pausedAt, pausedTotal;
function elapsed() {
  const now = state === "paused" ? pausedAt : performance.now();
  return now - startedAt - pausedTotal;
}

function start() {
  state = "playing"; x = 100; startedAt = performance.now();
  pausedAt = 0; pausedTotal = 0;
}

function update() {
  if (state !== "playing") return;
  x += 2; if (x > 700) x = 100;
  if (elapsed() >= 10000) state = "over";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "28px sans-serif";
  ctx.fillText("残り " + Math.max(0, Math.ceil((10000 - elapsed()) / 1000)) + "秒", 60, 80);
  ctx.fillRect(x, 200, 40, 40);
  ctx.fillText(state === "paused" ? "停止中　P: 再開" : state === "over" ? "時間切れ　R: 再開" : "P: 一時停止", 200, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
  if (event.key.toLowerCase() === "p") {
    if (state === "playing") { pausedAt = performance.now(); state = "paused"; }
    else if (state === "paused") { pausedTotal += performance.now() - pausedAt; state = "playing"; }
  }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();
