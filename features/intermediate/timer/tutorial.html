"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let startedAt, remaining;

function start() {
  startedAt = performance.now(); remaining = 10;
}

function update() {
  remaining = Math.max(0, 10 - (performance.now() - startedAt) / 1000);
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + Math.ceil(remaining) + "秒", 230, 160);

}



function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

開始時刻と現在時刻の差を秒に直し、10から引きます。0より小さくしません。

startedAt は開始時刻、remaining は残り秒です。現在時刻との差を1000で割るとミリ秒から秒になります。

保存後の確認

残り10秒から減ります。 この段階の見本を開く

小さな練習

表示に使う Math.ceil を Math.floor に変え、開始から2.5秒後には残り7秒と表示されることを確かめます。Math.ceilに戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let startedAt, remaining; let state;

function start() {
  startedAt = performance.now(); remaining = 10; state = "playing";
}

function update() {
  remaining = Math.max(0, 10 - (performance.now() - startedAt) / 1000);
  if (remaining === 0) state = "over";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + Math.ceil(remaining) + "秒", 230, 160);
  if (state === "over") ctx.fillText("時間切れ", 230, 220);

}



function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

残りが0のとき終了状態にします。別タブに移っても経過を含め、戻った次の更新で計算します。

remaining === 0 で state を over にします。Math.ceil は表示だけを切り上げます。

保存後の確認

10秒で時間切れになり、表示は0秒です。 この段階の見本を開く

小さな練習

終了判定の remaining === 0 を remaining <= 3 に変え、残り3秒で時間切れになることを確かめます。remaining === 0に戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let startedAt, remaining; let state;

function start() {
  startedAt = performance.now(); remaining = 10; state = "playing";
}

function update() {
  remaining = Math.max(0, 10 - (performance.now() - startedAt) / 1000);
  if (remaining === 0) state = "over";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + Math.ceil(remaining) + "秒", 230, 160);
  if (state === "over") ctx.fillText("時間切れ", 230, 220);
  ctx.fillText("R: 最初から", 230, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") start();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

Rで開始時刻を現在時刻にし、残りと状態を初期化します。OSのスリープ中に時間が進むかは環境によって異なります。

Rは start() を呼び、開始時刻を新しく記録します。毎回の update() は記録を変えず差を計算します。

保存後の確認

Rを押すと10秒から数え直します。 この段階の見本を開く

小さな練習

Rを調べる event.key.toLowerCase() === "r" のrをnに変え、Nを押すと残り10秒に戻ることを確かめます。rに戻します。

このステップ終了時の見本と自分のコードを比べる

自分のゲームに組み込むには

startedAt を開始と再挑戦のときだけ記録し、update() では現在時刻との差から remaining を求めます。毎回開始時刻を記録し直すと減らないため、初期化と更新を分けます。

見本の状態、入力、更新、描画を自分のゲームの名前に合わせて移します。すでにある start()、update()、draw() に必要な部分を加え、ゲームループは増やしません。やり直すときは追加した状態も初期化してください。
