コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

ひな型の index.html、style.css、script.js を作業用フォルダーへコピーします。変更前を探し、変更後のコードに置き換えて保存し、index.html を開き直します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, saved, keys;

function start() {
  x = 80; saved = 80; keys = {};
}

function update() {
  const old = x;
  x += (keys.ArrowRight ? 3 : 0) - (keys.ArrowLeft ? 3 : 0);
  x = Math.max(20, Math.min(780, x));

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#dce8ef"; ctx.fillRect(30, 190, 740, 70);
  ctx.fillStyle = "#3f9b68"; ctx.fillRect(398, 180, 4, 90);
  ctx.fillStyle = "#397ab4"; ctx.fillRect(x - 15, 210, 30, 30);
  ctx.fillStyle = "#26334a"; ctx.font = "22px sans-serif";
  ctx.fillText("位置: " + x + " / 戻る位置: " + saved, 100, 100);
  ctx.fillText("←→: 移動 / R: 戻る / N: 最初から", 100, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat && event.key.toLowerCase() !== "r") return;
  const key = event.key.toLowerCase();
  if (key === "n") { start(); return; }
  if (key === "r" && !event.repeat) { x = saved; keys = {}; return; }
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") { event.preventDefault(); keys[event.key] = true; }
});
window.addEventListener("keyup", event => { if (event.key === "ArrowLeft" || event.key === "ArrowRight") keys[event.key] = false; });
window.addEventListener("blur", () => { keys = {}; });

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, saved, keys;

function start() {
  x = 80; saved = 80; keys = {};
}

function update() {
  const old = x;
  x += (keys.ArrowRight ? 3 : 0) - (keys.ArrowLeft ? 3 : 0);
  x = Math.max(20, Math.min(780, x));
  if (old < 400 && x >= 400 && saved < 420) saved = 420;

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#dce8ef"; ctx.fillRect(30, 190, 740, 70);
  ctx.fillStyle = "#3f9b68"; ctx.fillRect(398, 180, 4, 90);
  ctx.fillStyle = "#397ab4"; ctx.fillRect(x - 15, 210, 30, 30);
  ctx.fillStyle = "#26334a"; ctx.font = "22px sans-serif";
  ctx.fillText("位置: " + x + " / 戻る位置: " + saved, 100, 100);
  ctx.fillText("←→: 移動 / R: 戻る / N: 最初から", 100, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat && event.key.toLowerCase() !== "r") return;
  const key = event.key.toLowerCase();
  if (key === "n") { start(); return; }
  if (key === "r" && !event.repeat) { x = saved; keys = {}; return; }
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") { event.preventDefault(); keys[event.key] = true; }
});
window.addEventListener("keyup", event => { if (event.key === "ArrowLeft" || event.key === "ArrowRight") keys[event.key] = false; });
window.addEventListener("blur", () => { keys = {}; });

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

const ctx = canvas.getContext("2d");

自分のゲームに組み込むには

x・savedと右向きの通過判定を移し、安全な場所をsavedに保存します。

自分のupdateで危険への接触後に位置と入力を戻します。得点や敵の状態も必要なら別に保存する設計が必要です。

gameLoopを増やさず、既存のstart・update・drawから処理を呼びます。再挑戦では持ち込んだ位置や状態、時刻を初期化します。
