記録の保存を作る

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

const storageKey = "game-example-best-v1";
let score, best, message;
function saveBest() {
  try { window.localStorage.setItem(storageKey, JSON.stringify({best})); message = "保存しました"; }
  catch (error) { message = "保存できません"; }
}
function loadBest() {
  try {
    const raw = window.localStorage.getItem(storageKey);
    if (raw === null) { message = "保存データがありません"; return; }
    const data = JSON.parse(raw);
    if (!data || !Number.isInteger(data.best) || data.best < 0 || data.best > 9999) { message = "保存データが不正です"; return; }
    best = data.best; message = "読み込みました";
  } catch (error) { message = "読み込めません"; }
}
function deleteBest() {
  try { window.localStorage.removeItem(storageKey); best = 0; message = "保存を消しました"; }
  catch (error) { message = "保存を消せません"; }
}

  score = 0; best = 0; message = "Spaceで得点";

  if (event.repeat) return;
  const key = event.key.toLowerCase();
  if (key === "r") { start(); return; }
  if (event.key === " ") { event.preventDefault(); score = Math.min(9999, score + 1); best = Math.max(best, score); }
  else if (key === "s") saveBest();
  else if (key === "l") loadBest();
  else if (key === "d") deleteBest();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();
