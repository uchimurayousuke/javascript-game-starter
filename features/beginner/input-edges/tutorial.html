ステップ 1：Spaceを押し始めたときだけ1回数える

目標

Spaceを押し始めたときだけ1回数える。

予想してみる

押し続けると何回数える？

答えを確認する

押し始めの1回です。

この段階で変えるところ

次は説明用の抜粋です。編集するときは、後の「比較とコピー用コード」を開き、変更後のコード全体を使います。

押した回数を用意する

let count;

function start() {
  count = 0;
}

count はSpaceを押した回数で、最初は0です。

押し始めだけ1回数える

if (event.repeat) return;
if (event.code === "Space") { count += 1; }

event.repeat が真なら処理を終えるので、押し続けている間の自動繰り返しでは増えません。離して押し直すと count をもう1増やします。Spaceの既定のスクロールは preventDefault() で防ぎます。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let count;

function start() {
  count = 0;
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("1回押し: " + count, 200, 120);
  ctx.fillText("Space: 数える", 100, 320);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space") event.preventDefault();
  if (event.repeat) return;
  if (event.code === "Space") { count += 1; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

ステップ 2：左右キーを押している間だけ毎更新3動く

目標

左右キーを押している間だけ毎更新3動く。

予想してみる

左右を同時に押すと？

答えを確認する

右と左が打ち消し合い動きません。

この段階で変えるところ

次は説明用の抜粋です。編集するときは、後の「比較とコピー用コード」を開き、変更後のコード全体を使います。

左右の状態を用意する

count = 0; x = 400; left = false; right = false;

x は四角の横位置、left と right は左右キーが押されているかを覚えます。開始時は横400、両方 false です。

押している間に動かす

x = Math.max(20, Math.min(780, x + (Number(right) - Number(left)) * 3));

Number(true) は1、Number(false) は0です。右だけなら1、左だけなら-1、両方なら0になり、3を掛けた分だけ位置を変えます。Math.min と Math.max で20から780に収めます。

押す・離す・画面から離れる

if (event.key === "ArrowLeft") { left = true; }
if (event.key === "ArrowRight") { right = true; }

押したキーを true にします。keyup は離したキーを false にし、blur はウィンドウが入力を受け取れなくなったとき両方を解除します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let count; let x, left, right;

function start() {
  count = 0; x = 400; left = false; right = false;
}

function update() {
  x = Math.max(20, Math.min(780, x + (Number(right) - Number(left)) * 3));
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("1回押し: " + count, 200, 120);
  ctx.fillRect(x - 15, 200, 30, 30);
  ctx.fillText("Space: 数える　左右: 移動", 100, 320);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space" || event.key === "ArrowLeft" || event.key === "ArrowRight") event.preventDefault();
  if (event.repeat) return;
  if (event.code === "Space") { count += 1; }
  if (event.key === "ArrowLeft") { left = true; }
  if (event.key === "ArrowRight") { right = true; }
});
window.addEventListener("keyup", event => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;
});
window.addEventListener("blur", () => { left = false; right = false; });

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

自分のゲームに組み込むには

押し始めで1回行う処理と、left・right を更新ごとに使う処理を分けます。keyup と blur で保持を解除し、再開始でも状態を消します。

見本の状態、入力、更新、描画を自分のゲームの名前に合わせて移します。すでにある start()、update()、draw() に必要な部分を加え、ゲームループは増やしません。やり直すときは追加した状態も初期化してください。

操作：Space:1回、左右:移動、R:最初から。
