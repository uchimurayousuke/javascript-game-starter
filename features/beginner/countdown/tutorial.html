開始前のカウントを作る

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let state, startedAt, x;

function start() {
  state = "ready"; startedAt = 0; x = 100;
}

function update() {
  if (state === "countdown" && performance.now() - startedAt >= 3000) state = "playing";
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "30px sans-serif";
  if (state === "ready") ctx.fillText("Enterで開始", 280, 170);
  else if (state === "countdown") ctx.fillText(String(Math.max(1, 3 - Math.floor((performance.now() - startedAt) / 1000))), 390, 170);
  else { ctx.fillText("スタート！", 300, 170); ctx.fillRect(x, 240, 40, 40); }
  ctx.fillText("R: はじめから", 280, 390);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
  if (event.key === "Enter" && state === "ready") { startedAt = performance.now(); state = "countdown"; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let state, startedAt, x;

function start() {
  state = "ready"; startedAt = 0; x = 100;
}

function update() {
  if (state === "countdown" && performance.now() - startedAt >= 3000) state = "playing";
  if (state === "playing") { x += 2; if (x > 700) x = 100; }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "30px sans-serif";
  if (state === "ready") ctx.fillText("Enterで開始", 280, 170);
  else if (state === "countdown") ctx.fillText(String(Math.max(1, 3 - Math.floor((performance.now() - startedAt) / 1000))), 390, 170);
  else { ctx.fillText("スタート！", 300, 170); ctx.fillRect(x, 240, 40, 40); }
  ctx.fillText("R: はじめから", 280, 390);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
  if (event.key === "Enter" && state === "ready") { startedAt = performance.now(); state = "countdown"; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

このステップ終了時の見本と自分のコードを比べる

自分のゲームに組み込むには

stateとstartedAt、カウント中の描画と時間判定を自作ゲームへ移します。

playingに切り替わる箇所でstartGameを呼び、四角の更新と描画を自作のupdateGame・drawGameに置き換えます。

この欄のstartGame・updateGame・drawGameは、自分のゲーム側の開始・更新・描画の関数名の例です。自分の関数名に合わせて読み替えます。gameLoopは増やさず、既存のstart・update・drawから必要な処理を呼び、再挑戦では持ち込んだ状態や入力も初期化します。
