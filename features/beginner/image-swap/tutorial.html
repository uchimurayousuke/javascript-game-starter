画像の差し替えを作る

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

変更 1：画像の読み込み状態を用意する

画像と状態を作る
const image = new Image();
const asset = {image, status: "loading"};

new Image() で画像を作り、画像と状態を asset にまとめます。最初の "loading" は読み込み中です。

読み込み結果を覚える
image.onload = () => { asset.status = image.naturalWidth > 0 && image.naturalHeight > 0 ? "ready" : "error"; };
image.onerror = () => { asset.status = "error"; };

読み込み完了時、元画像の幅と高さが両方0より大きければ "ready"、そうでなければ "error" にします。失敗時の onerror も "error" にします。&& は両方の条件が必要です。

読み込みを始める
image.src = src;
return asset;

完了と失敗の処理を決めてから src に画像の場所を入れ、asset を返します。読み込みはすぐに終わるとは限らないため、表示時に status を確かめます。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let images, selected;
function loadImage(src) {
  const image = new Image();
  const asset = {image, status: "loading"};
  image.onload = () => { asset.status = image.naturalWidth > 0 && image.naturalHeight > 0 ? "ready" : "error"; };
  image.onerror = () => { asset.status = "error"; };
  image.src = src;
  return asset;
}

function start() {
  selected = 0;
  images = [loadImage("assets/player-blue.png")];
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  const asset = images[selected];
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  if (asset.status === "ready") ctx.drawImage(asset.image, 300, 150);
  else ctx.fillText("画像を準備できません", 200, 100);

}

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let images, selected;
function loadImage(src) {
  const image = new Image();
  const asset = {image, status: "loading"};
  image.onload = () => { asset.status = image.naturalWidth > 0 && image.naturalHeight > 0 ? "ready" : "error"; };
  image.onerror = () => { asset.status = "error"; };
  image.src = src;
  return asset;
}

function start() {
  selected = 0;
  images = [loadImage("assets/player-blue.png"), loadImage("assets/player-alt.png")];
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  const asset = images[selected];
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  if (asset.status === "ready") ctx.drawImage(asset.image, 300, 150);
  else ctx.fillText("画像を準備できません", 200, 100);
  ctx.fillText("R: 読み直す / Space: 切替", 200, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
  if (event.key === " ") { event.preventDefault(); selected = 1 - selected; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let images, selected;
function loadImage(src) {
  const image = new Image();
  const asset = {image, status: "loading"};
  image.onload = () => { asset.status = image.naturalWidth > 0 && image.naturalHeight > 0 ? "ready" : "error"; };
  image.onerror = () => { asset.status = "error"; };
  image.src = src;
  return asset;
}

function start() {
  selected = 0;
  images = [loadImage("assets/player-blue.png"), loadImage("assets/player-alt.png")];
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  const asset = images[selected];
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  if (asset.status === "loading") ctx.fillText("画像を読み込み中", 200, 100);
  else if (asset.status === "error") ctx.fillText("画像を読み込めません。名前と場所を確認", 150, 100);
  else { ctx.drawImage(asset.image, 300, 150); ctx.fillText("画像を表示中", 200, 100); }
  ctx.fillText("R: 読み直す / Space: 切替", 200, 350);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }
  if (event.key === " ") { event.preventDefault(); selected = 1 - selected; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

画像を自分の絵に替えるには

stepまたはcompleteの index.html、style.css、script.js と assets フォルダーを一緒にコピーします。

同じ名前のPNGに自分の絵を保存すればコードはそのままで替えられます。別の名前や拡張子なら、script.jsの assets/player-blue.png や assets/player-alt.png を新しい名前に書き換えます。

assets/からの相対パス、名前の綴り、大文字と小文字、拡張子を確認してください。画像を読み込めない場合は表示せず、案内を出します。

自分のゲームに組み込むには

loadImageとimages、selectedを移し、自分のstartで読み込みます。

自分のdrawでstatusがreadyのときだけdrawImageを呼び、画像を切り替える入力を自分の操作へ接続します。

gameLoopを増やさず、既存のstart・update・drawから処理を呼びます。再挑戦では持ち込んだ位置や状態、時刻を初期化します。
