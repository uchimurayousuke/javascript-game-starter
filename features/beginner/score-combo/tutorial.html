"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let score;

function start() {
  score = 0;
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("得点: " + score, 250, 150);
  ctx.fillText("Space: 成功　", 160, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space") event.preventDefault();
  if (event.repeat) return;
  if (event.code === "Space") { score += 10; }
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

1回の成功で10を足します。キーを押し続けたときの繰り返し入力は受け取りません。

score が得点を覚え、score += 10 が元の得点に10を加えて戻します。

保存後の確認

Spaceを2回押すと得点20です。 この段階の見本を開く

小さな練習

score += 10 の10を5に変え、Spaceを2回押して得点10を確かめます。10に戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let score, combo;

function start() {
  score = 0; combo = 0;
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("得点: " + score, 250, 150);
  ctx.fillText("コンボ: " + combo, 250, 210);
  ctx.fillText("Space: 成功　F: 失敗　R: 最初から", 160, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.code === "Space") event.preventDefault();
  if (event.repeat) return;
  if (event.code === "Space") { combo += 1; score += 10 * combo; }
  if (event.key.toLowerCase() === "f") combo = 0;
  if (event.key.toLowerCase() === "r") start();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

Fはコンボだけを0にし、得点は残します。Rはstartを呼んで両方を初期化します。

combo = 0 は連続回数だけを戻します。start() は score と combo の両方を初期化します。

保存後の確認

Spaceを2回、F、Spaceの順に押すと得点40、コンボ1です。Rで0に戻ります。 この段階の見本を開く

小さな練習

Fの処理にある combo = 0 の0を1に変えます。Spaceを2回、F、Spaceと押すと得点50になることを確かめ、0に戻します。

このステップ終了時の見本と自分のコードを比べる

自分のゲームに組み込むには

score と combo を自分のゲームの状態に用意します。実際の成功判定でコンボを増やして加点し、失敗判定ではコンボだけを0にします。全体の再開始では両方を0にします。

見本の状態、入力、更新、描画を自分のゲームの名前に合わせて移します。すでにある start()、update()、draw() に必要な部分を加え、ゲームループは増やしません。やり直すときは追加した状態も初期化してください。

操作：Space:成功、F:失敗、R:最初から。
