"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let lives;

function start() {
  lives = 3;
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + lives, 250, 150);
  ctx.fillText("X: 失敗", 170, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.repeat) return;
  if (event.key.toLowerCase() === "x") lives = Math.max(0, lives - 1);
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

残り3から1を引き、Math.maxで0未満にならないようにします。

lives が残り回数です。Math.max(0, lives - 1) は引いた結果と0の大きい方を戻します。

保存後の確認

Xを1回押すと残り2です。 この段階の見本を開く

小さな練習

startの lives = 3 の3を2に変え、最初の残り2を確かめます。3に戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let lives, state;

function start() {
  lives = 3; state = "playing";
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + lives, 250, 150);
  if (state === "over") ctx.fillText("ゲームオーバー", 250, 220);
  ctx.fillText("X: 失敗　R: 最初から", 170, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.repeat) return;
  if (event.key.toLowerCase() === "x" && state === "playing") {
    lives = Math.max(0, lives - 1);
    if (lives === 0) state = "over";
  }
  if (event.key.toLowerCase() === "r") start();
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

残りが0になったら終了状態にします。終了中のXは受け付けず、Rで再開します。

state がプレイ中か終了かを覚えます。残り0になったとき over に変え、終了中の失敗処理を止めます。

保存後の確認

Xを3回押すとゲームオーバーです。Rで残り3に戻ります。 この段階の見本を開く

小さな練習

終了判定の lives === 0 の0を1に変え、Xを2回押すと残り1で終了することを確かめます。0に戻します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let lives, state;

function start() {
  lives = 3; state = "playing";
}

function update() {

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("残り: " + lives, 250, 150);
  if (state === "over") ctx.fillText("ゲームオーバー", 250, 220);
  ctx.fillText("X: 失敗　H: 回復　R: 最初から", 170, 300);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.repeat) return;
  if (event.key.toLowerCase() === "x" && state === "playing") {
    lives = Math.max(0, lives - 1);
    if (lives === 0) state = "over";
  }
  if (event.key.toLowerCase() === "r") start();
  if (event.key.toLowerCase() === "h" && state === "playing") lives = Math.min(3, lives + 1);
});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

コードの意味

回復するときもMath.minで上限3を守ります。終了後はRで最初から始めます。

Math.min(3, lives + 1) は増やした数と3の小さい方を戻します。終了中は回復処理も実行されません。

保存後の確認

Xの後にHで3に戻ります。3のときHを押しても増えません。 この段階の見本を開く

小さな練習

Hの処理にある Math.min(3, lives + 1) の3を4に変え、Hを押して残り4になることを確かめます。3に戻します。

このステップ終了時の見本と自分のコードを比べる

自分のゲームに組み込むには

lives と終了状態を自分のゲームに用意し、1回の失敗判定で1だけ減らします。キャラクターの位置だけを戻す処理と、全体の start() で残りを初期化する処理を分けます。

見本の状態、入力、更新、描画を自分のゲームの名前に合わせて移します。すでにある start()、update()、draw() に必要な部分を加え、ゲームループは増やしません。やり直すときは追加した状態も初期化してください。

操作：X:失敗、H:回復、R:最初から。
