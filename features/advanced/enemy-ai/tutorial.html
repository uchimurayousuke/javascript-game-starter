"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

class Enemy {
  constructor() { this.x = 160; this.direction = 1; this.state = "巡回"; }
  update(playerX) {
    this.x += this.direction * 2;
    if (this.x >= 320) { this.x = 320; this.direction = -1; }
    if (this.x <= 160) { this.x = 160; this.direction = 1; }
  }
}
let enemy, playerX, left, right;

function start() {
  enemy = new Enemy(); playerX = 480; left = false; right = false;
}

function update() {
  enemy.update(playerX);
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("敵: " + enemy.state, 180, 100);
  ctx.fillStyle = "#d85e59"; ctx.fillRect(enemy.x - 15, 200, 30, 30);
  ctx.fillStyle = "#3984c6"; ctx.fillRect(playerX - 15, 200, 30, 30);
  ctx.fillStyle = "#26334a"; ctx.fillText("敵が巡回", 150, 320);
}



function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

ステップ 2：左右キーでプレイヤーを動かす

目標

左右キーでプレイヤーを動かす。

予想してみる

右キーを放すと？

答えを確認する

右への移動が止まります。

小さな作業

上から順に作業します。目印を探し、その下のコードを書きます。

キーに応じて人を移動する

update() の敵を更新する行の上で、次の目印のすぐ上に追加します。

目印

  enemy.update(playerX);

追加するコード

  playerX = Math.max(20, Math.min(780, playerX + (Number(right) - Number(left)) * 3));

Number(true) は 1、Number(false) は 0 です。右だけなら 3 増え、左だけなら 3 減ります。Math.min と Math.max で位置を 20 から 780 に収めます。

人の操作を案内する

draw() の巡回案内で、次の範囲を置き換えます。

置き換える範囲

  ctx.fillStyle = "#26334a"; ctx.fillText("敵が巡回", 150, 320);

置き換えるコード

  ctx.fillStyle = "#26334a"; ctx.fillText("左右: 移動　R: 最初から", 150, 320);

左右で移動、R で再開する操作を表示します。

キー入力を受け取る

小文字にしたキーが R なら start() で位置と入力を初期化します。

キーを離したことを記録する

keydown の処理の後で、次の目印のすぐ下に追加します。

目印

  if (event.key.toLowerCase() === "r") start();
});

追加するコード

window.addEventListener("keyup", event => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;

keyup はキーを離したときに起き、対応する値を false に戻します。

画面から離れたら入力を解除する

window.addEventListener("blur", () => { left = false; right = false; });

別の画面へ移ると左右の記録を消します。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

class Enemy {
  constructor() { this.x = 160; this.direction = 1; this.state = "巡回"; }
  update(playerX) {
    this.x += this.direction * 2;
    if (this.x >= 320) { this.x = 320; this.direction = -1; }
    if (this.x <= 160) { this.x = 160; this.direction = 1; }
  }
}
let enemy, playerX, left, right;

function start() {
  enemy = new Enemy(); playerX = 480; left = false; right = false;
}

function update() {
  playerX = Math.max(20, Math.min(780, playerX + (Number(right) - Number(left)) * 3));
  enemy.update(playerX);
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("敵: " + enemy.state, 180, 100);
  ctx.fillStyle = "#d85e59"; ctx.fillRect(enemy.x - 15, 200, 30, 30);
  ctx.fillStyle = "#3984c6"; ctx.fillRect(playerX - 15, 200, 30, 30);
  ctx.fillStyle = "#26334a"; ctx.fillText("左右: 移動　R: 最初から", 150, 320);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") event.preventDefault();
  if (event.repeat) return;
  if (event.key === "ArrowLeft") { left = true; }
  if (event.key === "ArrowRight") { right = true; }
  if (event.key.toLowerCase() === "r") start();
});
window.addEventListener("keyup", event => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;
});
window.addEventListener("blur", () => { left = false; right = false; });

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

class Enemy {
  constructor() { this.x = 160; this.direction = 1; this.state = "巡回"; }
  update(playerX) {
    this.x += this.direction * 2;
    if (this.x >= 320) { this.x = 320; this.direction = -1; }
    if (this.x <= 160) { this.x = 160; this.direction = 1; }
  }
}
let enemy, playerX, left, right;

function start() {
  enemy = new Enemy(); playerX = 480; left = false; right = false;
}

function update() {
  playerX = Math.max(20, Math.min(780, playerX + (Number(right) - Number(left)) * 3));
  enemy.update(playerX);
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a";
  ctx.font = "28px sans-serif";
  ctx.fillText("敵: " + enemy.state, 180, 100);
  ctx.fillStyle = "#d85e59"; ctx.fillRect(enemy.x - 15, 200, 30, 30);
  ctx.fillStyle = "#3984c6"; ctx.fillRect(playerX - 15, 200, 30, 30);
  ctx.fillStyle = "#26334a"; ctx.fillText("左右: 移動　R: 最初から", 150, 320);
}

window.addEventListener("keydown", event => {
  if (event.ctrlKey || event.altKey || event.metaKey || event.isComposing) return;
  if (event.key === "ArrowLeft" || event.key === "ArrowRight") event.preventDefault();
  if (event.repeat) return;
  if (event.key === "ArrowLeft") { left = true; }
  if (event.key === "ArrowRight") { right = true; }
  if (event.key.toLowerCase() === "r") start();
});
window.addEventListener("keyup", event => {
  if (event.key === "ArrowLeft") left = false;
  if (event.key === "ArrowRight") right = false;
});
window.addEventListener("blur", () => { left = false; right = false; });

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

class Enemy {
  constructor() { this.x = 160; this.direction = 1; this.state = "巡回"; }
  update(playerX) {
    if (Math.abs(playerX - this.x) <= 100) {
      this.state = "追跡";
      const difference = playerX - this.x;
      this.x += Math.sign(difference) * Math.min(2, Math.abs(difference));
      return;
    }
    this.state = "巡回";
    if (this.x > 320) { this.x = Math.max(320, this.x - 2); this.direction = -1; return; }
    if (this.x < 160) { this.x = Math.min(160, this.x + 2); this.direction = 1; return; }
    this.x += this.direction * 2;
    if (this.x >= 320) { this.x = 320; this.direction = -1; }
    if (this.x <= 160) { this.x = 160; this.direction = 1; }
  }
}

自分のゲームに組み込むには

new Enemy() で敵を作り、更新時に enemy.update(player.x) のようにプレイヤー位置を渡します。敵とプレイヤーは同じ世界座標で距離を比べます。範囲外からは巡回端まで少しずつ戻します。

見本の状態、入力、更新、描画を自分のゲームの名前に合わせて移します。すでにある start()、update()、draw() に必要な部分を加え、ゲームループは増やしません。やり直すときは追加した状態も初期化してください。

操作：左右:移動、R:最初から。

敵の追跡を自作へ移すときは、巡回範囲の外からも急に移さず、端に少しずつ戻す処理を合わせて使います。
