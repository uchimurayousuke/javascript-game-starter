時間を使った移動を作る

コードの比較の読み方

比較欄では消す行、追加する行、そのままの行を色とラベルで示します。変更のない途中の行を省略することがあります。比較欄は読むためのもので、コピーする欄ではありません。

− 消す＋ 追加そのまま

置き換えるときは、赤い行だけを消すのではなく、指定された変更前の範囲すべてをコピー用のコードに置き換えます。そのままの行もコピー用のコードに含まれます。古い関数を残して同じ名前の関数を追加しないでください。

コピーするのは「コピー用」と書かれたコード欄です。比較の記号はコードに入りません。

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, previous, delta;

function start() {
  x = 100; previous = null; delta = 0;
}

function update() {
  const now = performance.now();
  if (previous === null) { previous = now; delta = 0; return; }
  delta = Math.max(0, (now - previous) / 1000); previous = now;

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  ctx.fillText("1回の経過秒: " + delta.toFixed(3), 100, 100);
  ctx.fillText("x: " + x.toFixed(1) + " / 速さ: 120px/秒", 100, 150);
  ctx.fillRect(x - 20, 220, 40, 40);
  ctx.fillText("R: 最初から", 100, 360);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }

});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, previous, delta;

function start() {
  x = 100; previous = null; delta = 0;
}

function update() {
  const now = performance.now();
  if (previous === null) { previous = now; delta = 0; return; }
  delta = Math.max(0, (now - previous) / 1000); previous = now;
  x += 120 * delta;
  if (x > 800) x = 0;

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  ctx.fillText("1回の経過秒: " + delta.toFixed(3), 100, 100);
  ctx.fillText("x: " + x.toFixed(1) + " / 速さ: 120px/秒", 100, 150);
  ctx.fillRect(x - 20, 220, 40, 40);
  ctx.fillText("R: 最初から", 100, 360);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }

});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

変更 1

変更前（確認用）：

  delta = Math.max(0, (now - previous) / 1000); previous = now;

Math.min で1回の経過秒を最大0.05にします。長い停止後に一度で遠くへ飛ぶのを防ぎ、移動は最大 120 × 0.05 = 6 ピクセルです。上限を超えた時間は移動に使いません。1/30秒の更新30回も1/60秒の更新60回も、合計1秒なら120ピクセルです。

この段階の script.js 全文を確認する
"use strict";

const canvas = document.querySelector("#game-canvas");
const ctx = canvas.getContext("2d");

let x, previous, delta;

function start() {
  x = 100; previous = null; delta = 0;
}

function update() {
  const now = performance.now();
  if (previous === null) { previous = now; delta = 0; return; }
  delta = Math.max(0, (now - previous) / 1000); previous = now;
  delta = Math.min(delta, 0.05);
  x += 120 * delta;
  if (x > 800) x = 0;

}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#26334a"; ctx.font = "24px sans-serif";
  ctx.fillText("1回の経過秒: " + delta.toFixed(3), 100, 100);
  ctx.fillText("x: " + x.toFixed(1) + " / 速さ: 120px/秒", 100, 150);
  ctx.fillRect(x - 20, 220, 40, 40);
  ctx.fillText("R: 最初から", 100, 360);
}

window.addEventListener("keydown", event => {
  if (event.repeat) return;
  if (event.key.toLowerCase() === "r") { start(); return; }

});

function gameLoop() {
  update();
  draw();
  requestAnimationFrame(gameLoop);
}

start();
gameLoop();

自分のゲームに組み込むには

previousとdelta、経過時間の計算を自分のupdateへ移します。

startや再開時にpreviousをnullにし、1回のdeltaを制限して、速さ×deltaを位置へ足します。

gameLoopを増やさず、既存のstart・update・drawから処理を呼びます。再挑戦では持ち込んだ位置や状態、時刻を初期化します。
